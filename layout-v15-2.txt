    rirInput.setAttribute('aria-label', `Target RIR for ${ex.name}`);
    rirInput.addEventListener('input', () => { ex.targetRir = rirInput.value; saveData(); });
    rir.appendChild(rirInput);
    row.appendChild(rir);

    const repAim = document.createElement('td');
    repAim.className = 'sheet-rep-aim-cell';
    repAim.textContent = `${cfg.min}-${cfg.max}`;
    row.appendChild(repAim);

    const control = document.createElement('td');
    control.className = 'sheet-control-cell';
    const controls = document.createElement('div');
    controls.className = 'sheet-set-controls';
    const add = document.createElement('button');
    add.className = 'sheet-set-control'; add.textContent = '+'; add.title = 'Add set';
    const remove = document.createElement('button');
    remove.className = 'sheet-set-control'; remove.textContent = '−'; remove.title = 'Remove last set';
    remove.disabled = sets.length <= 1;
    add.addEventListener('click', () => {
      sets.push({weight:'', reps:'', failure:false, dropset:false});
      ex.defaultSets = Math.max(ex.defaultSets || 1, sets.length);
      saveData();
      renderWorkout(day.id);
    });
    remove.addEventListener('click', () => {
      if (sets.length <= 1) return;
      const last = sets[sets.length - 1];
      const entered = String(last.weight ?? '').trim() || String(last.reps ?? '').trim() || last.failure || last.dropset;
      if (entered && !confirm('Remove the last set and its entered data?')) return;
      sets.pop();
      ex.defaultSets = Math.min(ex.defaultSets || sets.length, sets.length);
      saveData();
      renderWorkout(day.id);
    });
    controls.append(add, remove);
    control.appendChild(controls);
    row.appendChild(control);
    return row;
  }
