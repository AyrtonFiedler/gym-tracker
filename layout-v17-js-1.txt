  function finishV17(dayId){
    const day=getDay(dayId); if(!day)return;
    const log=dayLog(dayId); const has=day.exercises.some(ex=>completedSets(log[ex.id]||[]).length>0); if(!has){alert('Enter at least one completed working set first.');return;}
    if(!data.history[dayId])data.history[dayId]={}; const stamp=new Date().toISOString();
    day.exercises.forEach(ex=>{
      const sets=log[ex.id]||[], completed=completedSets(sets), warm=warmupFor(dayId,ex.id);
      if(completed.length){
        if(!data.history[dayId][ex.id])data.history[dayId][ex.id]=[];
        const ww=Number(warm.weight),rr=Number(warm.reps); const warmup=Number.isFinite(ww)&&ww>=0&&Number.isFinite(rr)&&rr>0?{weight:ww,reps:rr}:null;
        data.history[dayId][ex.id].push({date:stamp,warmup,sets:completed.map(s=>({weight:s.weight,reps:s.reps,failure:s.failure,dropset:s.dropset}))});
        if(data.history[dayId][ex.id].length>30)data.history[dayId][ex.id].shift();
      }
      log[ex.id]=Array.from({length:Math.max(1,ex.defaultSets||completed.length||sets.length||1)},emptySet); warmupsForDay(dayId)[ex.id]={weight:'',reps:''};
    });
    saveData(); showWorkout(dayId);
  }
