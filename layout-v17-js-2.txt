  function deleteExercise(dayId,exId){
    const day=getDay(dayId),ex=getEx(day,exId); if(!day||!ex)return; if(!confirm(`Delete ${ex.name}?`))return; day.exercises=day.exercises.filter(e=>e.id!==exId); if(data.logs[dayId])delete data.logs[dayId][exId]; if(data.history[dayId])delete data.history[dayId][exId]; if(data.warmups[dayId])delete data.warmups[dayId][exId]; saveData(); showWorkout(dayId);
  }

  function sessionVolume(session){return (session?.sets||[]).reduce((sum,s)=>{const w=Number(s.weight),r=Number(s.reps);return sum+(Number.isFinite(w)&&Number.isFinite(r)?w*r:0);},0);}
  function bestSet(session){
    const sets=completedSets(session?.sets||[]).filter(s=>!s.dropset); if(!sets.length)return null; return sets.slice().sort((a,b)=>{const ea=estimatedOneRepMax(a.weight,a.reps)||0,eb=estimatedOneRepMax(b.weight,b.reps)||0;return eb-ea;})[0];
  }
  function allSets(dayId,exId){ return historyFor(dayId,exId).flatMap(s=>(s.sets||[]).map(x=>({...x,date:s.date}))); }
  function prs(dayId,exId){
    const hist=historyFor(dayId,exId),sets=allSets(dayId,exId).filter(s=>Number(s.reps)>0&&Number(s.weight)>=0); if(!sets.length)return null;
    const topWeight=sets.slice().sort((a,b)=>Number(b.weight)-Number(a.weight)||Number(b.reps)-Number(a.reps))[0]; const mostReps=sets.slice().sort((a,b)=>Number(b.reps)-Number(a.reps)||Number(b.weight)-Number(a.weight))[0]; const e=sets.map(s=>({s,v:estimatedOneRepMax(Number(s.weight),Number(s.reps))||0})).sort((a,b)=>b.v-a.v)[0]; return {topWeight,mostReps,e1rm:e,totalVolume:hist.reduce((a,s)=>a+sessionVolume(s),0),sessions:hist.length};
  }
  function simpleLineChart(points){
    const W=330,H=145,L=30,R=8,T=8,B=24;if(!points.length)return'<div class="v17-subtle" style="padding:25px;text-align:center">No chart data yet.</div>'; const vals=points.map(p=>p.v);let min=Math.min(...vals),max=Math.max(...vals);if(max-min<1){min=Math.max(0,min-1);max+=1}else{const pad=(max-min)*.12;min=Math.max(0,min-pad);max+=pad;} const x=i=>points.length===1?L+(W-L-R)/2:L+i*(W-L-R)/(points.length-1);const y=v=>T+(max-v)*(H-T-B)/(max-min);let s=`<svg class="v17-chart" viewBox="0 0 ${W} ${H}">`;for(let i=0;i<4;i++){const yy=T+i*(H-T-B)/3;s+=`<line class="v17-chart-grid" x1="${L}" y1="${yy}" x2="${W-R}" y2="${yy}"/>`;}s+=`<polyline class="v17-chart-line" points="${points.map((p,i)=>`${x(i)},${y(p.v)}`).join(' ')}"/>`;points.forEach((p,i)=>s+=`<circle class="v17-chart-dot" cx="${x(i)}" cy="${y(p.v)}" r="3"/>`);const labels=[0,Math.floor((points.length-1)/2),points.length-1];[...new Set(labels)].forEach(i=>{s+=`<text class="v17-chart-label" x="${x(i)}" y="${H-6}" text-anchor="middle">${esc(formatDateShort(points[i].date))}</text>`});s+='</svg>';return s;
  }
